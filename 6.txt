mysql> SELECT customer_id, total
     FROM (
     SELECT Orders.customer_id,
     SUM(Product.price * Orders.quantity) AS total
     FROM Orders
     JOIN Product ON Orders.product_id = Product.product_id
     GROUP BY Orders.customer_id
     ) AS Purchase
     WHERE total > 30000;
