mysql> DELIMITER //
mysql> CREATE PROCEDURE total_marks_proc()
BEGIN
DECLARE done INT DEFAULT 0;
DECLARE m INT;
DECLARE total INT DEFAULT 0;
DECLARE cur CURSOR FOR
SELECT marks FROM student;
DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;
OPEN cur;
read_loop: LOOP
FETCH cur INTO m;
IF done = 1 THEN
LEAVE read_loop;
END IF;
SET total = total + m;
END LOOP;
CLOSE cur;
SELECT total AS Total_Marks;
END //
Query OK, 0 rows affected (0.02 sec)
