x. Show number of orders placed by each customer
db.orders.aggregate([ { $group: { _id: "$Cust_name", cnt_of_order : { $sum: 1 } } } ])

xi. Display customer IDs and total pending order amount in descending order.
db.orders.aggregate([
  { $match: { Status: "P" } },
  { $group: { _id: "$Cust_id", pend_amt: { $sum: "$Amt" } } },
  { $sort: { pend_amt: -1 } }
])

xii. Display customer IDs in ascending order with total delivered order amount
db.orders.aggregate([
  { $match: { Status: "D" } },
  { $group: { _id: "$Cust_id", tot_amt: { $sum: "$Amt" } } },
  { $sort: { _id: 1 } }
])

xiii. Show top two selling items
db.orders.aggregate([
  { $group: { _id: "$Item_name", totqty: { $sum: "$Qty" } } },
  { $sort: { totqty: -1 } },
  { $limit: 2 }
])

xiv. Find date on which maximum orders received
